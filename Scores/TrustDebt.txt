# trust_debt.py — implements §4.5
# TD^(p)(t) = sum( max(0, θ^(p)(t) - T_x^(p)(t)) ) for all x in X_p

def trust_debt_phase(theta: float, trust_scores: list) -> float:
    '''
    §4.5: TD^(p) = sum(max(0, θ - T_x)) for all x in X_p
    Bounded: 0 ≤ TD ≤ |X_p|
    '''
    assert 0 <= theta <= 1
    assert all(0 <= t <= 1 for t in trust_scores)
    TD = sum(max(0.0, theta - T_x) for T_x in trust_scores)
    assert 0 <= TD <= len(trust_scores), f'Bound violated: TD={TD}'
    return round(TD, 4)

    return round(sum(phase_debts.values()), 4)
