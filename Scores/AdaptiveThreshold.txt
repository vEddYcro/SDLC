# adaptive_threshold.py — implements §4.3
# θ^(p)(t+1) = clip(θ^(p)(t) + α·R(t))

def clip(z: float) -> float:
    '''clip(z) = min(1, max(0, z)) as defined in §4.3'''
    return min(1.0, max(0.0, z))

def adapt_threshold(theta_t: float, alpha: float, R_t: float) -> float:
    '''
    §4.3: θ^(p)(t+1) = clip(θ^(p)(t) + α·R(t))
    Guarantees: 0 ≤ θ(t+1) ≤ 1
    '''
    assert 0 <= theta_t <= 1, f'theta_t must be in [0,1], got {theta_t}'
    assert 0 <= alpha  <= 1, f'alpha must be in [0,1], got {alpha}'
    assert 0 <= R_t    <= 1, f'R_t must be in [0,1], got {R_t}'
    theta_next = clip(theta_t + alpha * R_t)
    assert 0 <= theta_next <= 1, 'Boundedness invariant violated'
    return round(theta_next, 4)

def check_gate(T_x: float, theta: float) -> bool:
    '''§4.3: component may progress only if T_x >= θ'''
    return T_x >= theta
