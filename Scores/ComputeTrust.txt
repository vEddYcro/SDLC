#!/usr/bin/env python3
# compute_trust.py — implements §4.2 T_x^(p) = sum(w_i * V_x_i)

# Phase weight vectors — must sum to 1.0 per phase
# Format: [w_I, w_C, w_B, w_H, w_E]
WEIGHTS = {
    'p1_planning':     [0.30, 0.20, 0.10, 0.20, 0.20],
    'p2_development':  [0.25, 0.25, 0.25, 0.15, 0.10],
    'p3_build':        [0.15, 0.25, 0.30, 0.20, 0.10],
    'p4_release':      [0.20, 0.20, 0.20, 0.20, 0.20],
    'p5_operate':      [0.20, 0.15, 0.25, 0.15, 0.25],
}

def compute_trust(phase: str, V: list) -> float:
    w = WEIGHTS[phase]
    assert abs(sum(w) - 1.0) < 1e-9, f'Weights must sum to 1, got {sum(w)}'
    assert all(0 <= v <= 1 for v in V), f'All V_i must be in [0,1], got {V}'
    T = sum(w[i] * V[i] for i in range(5))
    assert 0 <= T <= 1, f'Invariant violated: T={T} not in [0,1]'
    return round(T, 4)
