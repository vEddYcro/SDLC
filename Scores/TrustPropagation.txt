#!/usr/bin/env python3
# trust_propagation.py — implements §4.4 Gödel t-norm propagation

from typing import Dict, List

class ArtifactTrustDAG:
    '''Implements trust propagation over DAG per §4.4'''

    def __init__(self):
        self.nodes: Dict[str, float] = {}
        self.edges: Dict[str, List[str]] = {}

    def add_artifact(self, name: str, local_trust: float):
        assert 0 <= local_trust <= 1, f'{name}: trust must be in [0,1]'
        self.nodes[name] = local_trust
        self.edges.setdefault(name, [])

    def add_dependency(self, artifact: str, depends_on: str):
        self.edges.setdefault(artifact, []).append(depends_on)

    def propagated_trust(self, artifact: str, visited=None) -> float:
        '''§4.4: T_a = min(local, min(T_deps)) — Gödel t-norm'''
        if visited is None:
            visited = set()
        assert artifact not in visited, f'Cycle detected at {artifact}'
        visited.add(artifact)
        deps = self.edges.get(artifact, [])
        if not deps:
            return self.nodes[artifact]
        dep_trusts = [self.propagated_trust(d, visited.copy()) for d in deps]
        return round(min(self.nodes[artifact], min(dep_trusts)), 4)

    def assert_no_amplification(self):
        '''§4.4 invariant: propagated trust <= local trust for all nodes'''
        violations = []
        for art in self.nodes:
            prop = self.propagated_trust(art)
            local = self.nodes[art]
            if prop > local + 1e-9:
                violations.append(f'{art}: prop={prop} > local={local}')
        if violations:
            print(f'FAIL — amplification detected: {violations}')
        else:
            print('PASS — No trust amplification through DAG propagation ✓')
